OCP Concept

Open/Closed Principle (OCP)

বাস্তব উদাহরণ: দেয়ালের সকেট। বোর্ড না ভেঙেই নতুন চার্জার বা ল্যাপটপ প্লাগ-ইন করতে পারেন।

If-Else Blocked
সরাসরি মডিফিকেশনের ঝুঁকি
নতুন কোনো ডিসকাউন্ট লজিক যোগ করতে হলে আপনাকে ক্যালকুলেট ক্লাসে ঢুকে if কন্ডিশন বাড়াতে হবে, যা OCP ভঙ্গ করে।
public class InvoiceCalculator
{
    public double Calculate(double amt, string type)
    {
        // ❌ ডিসকাউন্ট যোগ করতে কোড এডিট করতে হচ্ছে
        if (type == "Regular") return amt * 0.9;
        else if (type == "VIP") return amt * 0.8;
        return amt;
    }
}
public interface IDiscount { double Apply(double amt); }

public class RegularDiscount : IDiscount {
    public double Apply(double amt) => amt * 0.9;
}
public class VIPDiscount : IDiscount {
    public double Apply(double amt) => amt * 0.8;
}

// ✅ ওটিপি ক্যালকুলেটর (নতুন ক্লাস বাড়িয়ে পলিসি এক্সটেন্ড করা সম্ভব)
public class InvoiceCalculator {
    public double Calculate(double amt, IDiscount discount) =>
        discount.Apply(amt);
}

public class Bird {
    public virtual void Fly() { }
}

    public override void Fly() =>
        throw new NotImplementedException();
}
public class Bird { }

public interface IFlyingBird { void Fly(); }

public class Sparrow : Bird, IFlyingBird {
    public void Fly() { }
}

public class Ostrich : Bird {
    // ✅ সেফ ডিজাইন, জোরপূর্বক উড়ার চুক্তি নেই
}

Fat Interface
অনাবশ্যক কোডের ভার

public interface IMultiFunctionPrinter {
    void Print();
    void Scan();
    void Fax();
}

public class BasicPrinter : IMultiFunctionPrinter {
    public void Print() { }
    public void Scan() => throw new NotSupportedException();
    public void Fax() => throw new NotSupportedException();
}
public interface IPrinter { void Print(); }
public interface IScanner { void Scan(); }
public interface IFax { void Fax(); }

public class BasicPrinter : IPrinter {
    public void Print() { }

DIP Concept

Dependency Inversion Principle (DIP)

বাস্তব উদাহরণ: একটি সিলিং লাইট সরাসরি পাওয়ার লাইনে সোল্ডারিং না করে সকেটের সাথে যুক্ত করা। সকেট হলো ইন্টারফেস (Abstraction)।

Tight Coupling
কংক্রিট ম্যাপের ঝুঁকি
StoreService সরাসরি SqlDatabase কংক্রিট ক্লাসে নির্ভরশীল। ডাটাবেজ বদলালে সার্ভিস রি-বিল্ড করতে হবে।
public class SqlDatabase {
    public void Save(string data) { }
}

public class StoreService {
    // ❌ SQL কাস্টমাইজেশন ছাড়া সার্ভিস অচল
    private SqlDatabase _db = new SqlDatabase();
    public void SaveData(string d) => _db.Save(d);
}
public interface IDatabase { void Save(string data); }

public class StoreService {
    private readonly IDatabase _db;
    // ✅ ইন্টারফেসের ওপর নির্ভরশীল StoreService
    public StoreService(IDatabase db) { _db = db; }
    public void SaveData(string d) => _db.Save(d);
}
