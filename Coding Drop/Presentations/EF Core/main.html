AsNoTracking — গোয়েন্দা বন্ধ করুন
Tracking (Default)
AsNoTracking
// ❌ Tracking ON — Read-only কাজে unnecessary overhead
public async Task<List<Product>> GetProductsAsync() {
    // Change Tracker সব Product মনে রাখছে!
    // Memory বেশি নেয়, Performance কম
    return await _db.Products
        .Where(p => p.IsActive)
        .ToListAsync();
}
// ✅ AsNoTracking — Read-only query-তে
public async Task<List<Product>> GetProductsAsync() {
    return await _db.Products
        .AsNoTracking()     // ✅ Tracker বন্ধ
        .Where(p => p.IsActive)
        .ToListAsync();
    // ৩০-৫০% দ্রুত! Update করবো না যখন
}

Include — Eager Loading
Lazy (N+1)
Include (সঠিক)
// ❌ Lazy Loading — N+1 Query সমস্যা
var customers = await _db.Customers.ToListAsync();

foreach (var customer in customers) {
    // প্রতিটি customer-এর জন্য আলাদা DB Query!
    var orders = customer.Orders; // ← এখানে DB Hit!
    Console.WriteLine(orders.Count);
}
// ১০০ Customer = ১০১ DB Query 😱
// ✅ Eager Loading — একটি JOIN Query
var customers = await _db.Customers
    .Include(c => c.Orders)           // JOIN করো
        .ThenInclude(o => o.OrderItems) // Nested join
    .AsNoTracking()                   // Read-only
    .ToListAsync();

foreach (var c in customers) {
    Console.WriteLine(c.Orders.Count); // DB Hit নেই!
}
// ১০০ Customer = মাত্র ১ DB Query ✅

// ❌ সব Column SELECT — অপ্রয়োজনীয় data transfer
var products = await _db.Products
    .ToListAsync(); // SELECT * FROM Products
// হাজার column, কিন্তু শুধু Id এবং Name দরকার ছিলো
// ✅ Select — শুধু দরকারি field আনুন
var products = await _db.Products
    .Where(p => p.IsActive)
    .Select(p => new ProductDto {
        Id = p.Id,
        Name = p.Name,
        Price = p.Price  // শুধু ৩টি field!
    })
    .AsNoTracking()
    .ToListAsync(); // SELECT Id, Name, Price — দ্রুত!
Compiled Queries — Hot Path অপ্টিমাইজ
// ✅ Compiled Query — LINQ এক্সপ্রেশন cache করে
private static readonly Func<AppDbContext, int, Task<User?>>
    _getUserById = EF.CompileAsyncQuery(
        (AppDbContext db, int id) =>
            db.Users.FirstOrDefault(u => u.Id == id));

// Usage — Expression Tree compile আর করতে হয় না
public async Task<User?> GetUserAsync(int id) {
    return await _getUserById(_db, id);
    // প্রথম call-এর পর compile overhead নেই!
    // High-traffic API-তে ব্যবহার করুন
}
Bulk Operations (.NET 7+)
Loop Update (ধীর)
ExecuteUpdate (দ্রুত)
// ❌ Loop — প্রতিটি Entity Track করে, N UPDATE Query
var products = await _db.Products
    .Where(p => p.CategoryId == categoryId)
    .ToListAsync();   // সব DB থেকে আনলো

foreach (var p in products)
    p.IsActive = false; // Change Tracker ট্র্যাক করছে

await _db.SaveChangesAsync(); // N UPDATE Queries!
// ✅ ExecuteUpdateAsync — একটি SQL UPDATE
await _db.Products
    .Where(p => p.CategoryId == categoryId)
    .ExecuteUpdateAsync(s => s
        .SetProperty(p => p.IsActive, false));
// UPDATE Products SET IsActive=0 WHERE CategoryId=@id
// একটি Query! Memory-তে কিছু আনতে হয়নি!
