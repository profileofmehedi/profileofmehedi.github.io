Domain Layer — Entity & Interface
// Domain/Entities/Order.cs — শুধু Business Logic
public class Order : BaseEntity {
    public Money TotalAmount { get; private set; }
    public OrderStatus Status { get; private set; }
    public List<OrderItem> Items { get; private set; }

    public void AddItem(Product product, int qty) {
        if (Status != OrderStatus.Draft)
            throw new OrderNotEditableException(); // Domain Rule
        Items.Add(new OrderItem(product, qty));
        TotalAmount = CalculateTotal();
    }
}

// Domain/Interfaces/IOrderRepository.cs — Interface only
public interface IOrderRepository {
    Task<Order?> GetByIdAsync(Guid id);
    Task AddAsync(Order order);
}

MediatR — Command Handler
// Application/Features/Orders/CreateOrderCommand.cs
public record CreateOrderCommand(string CustomerId) : IRequest<Guid>;

// Application/Features/Orders/CreateOrderCommandHandler.cs
public class CreateOrderCommandHandler
    : IRequestHandler<CreateOrderCommand, Guid> {

    private readonly IOrderRepository _repo; // Interface!
    private readonly IUnitOfWork _uow;

    public async Task<Guid> Handle(
        CreateOrderCommand request, CancellationToken ct) {
        var order = new Order(request.CustomerId); // Domain Object
        await _repo.AddAsync(order);
        await _uow.SaveChangesAsync(ct);
        return order.Id;
    }
}

// ❌ Fat Controller — Business Logic Controller-এ
public async Task<IActionResult> CreateOrder([FromBody] CreateOrderDto dto) {
    if (!ModelState.IsValid) return BadRequest();
    var customer = await _db.Customers.FindAsync(dto.CustomerId);
    if (customer == null) return NotFound();
    // 50 লাইন Business Logic... ❌
    var order = new Order { CustomerId = dto.CustomerId };
    _db.Orders.Add(order);
    await _db.SaveChangesAsync();
    return Ok(order.Id);
}
// ✅ Thin Controller — শুধু MediatR-এ পাঠায়
[HttpPost]
public async Task<IActionResult> CreateOrder(
    [FromBody] CreateOrderDto dto) {
    var command = new CreateOrderCommand(dto.CustomerId);
    var orderId = await _mediator.Send(command);
    return CreatedAtAction(nameof(GetOrder), new { id = orderId }, null);
}

// Domain/Interfaces/IOrderRepository.cs
public interface IOrderRepository {
    Task<Order?> GetByIdAsync(Guid id, CancellationToken ct = default);
    Task<List<Order>> GetByCustomerIdAsync(string customerId);
    Task AddAsync(Order order);

// Infrastructure/Repositories/OrderRepository.cs
public class OrderRepository : IOrderRepository {
    private readonly AppDbContext _db;

    public async Task<Order?> GetByIdAsync(Guid id, CancellationToken ct) {
        return await _db.Orders
            .Include(o => o.Items)
            .FirstOrDefaultAsync(o => o.Id == id, ct);
    }
}

// Infrastructure/DependencyInjection.cs
public static class InfrastructureServiceExtensions {
    public static IServiceCollection AddInfrastructure(
        this IServiceCollection services, IConfiguration config) {

        services.AddDbContext<AppDbContext>(opts =>
            opts.UseSqlServer(config.GetConnectionString("Default")));

        services.AddScoped<IOrderRepository, OrderRepository>();
        services.AddScoped<IUnitOfWork, UnitOfWork>();
        services.AddScoped<IEmailService, SmtpEmailService>();
        return services;
    }
}

public class CreateOrderCommandHandlerTests {

    [Fact]
    public async Task Handle_ValidCommand_ReturnsOrderId() {
