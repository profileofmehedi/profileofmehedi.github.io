// ❌ Synchronous — Thread ব্লক হয়ে থাকে
public IActionResult GetUser(int id) {
    var user = _db.Users.Find(id); // Thread আটকে আছে!
    var orders = _orderService.GetOrders(id); // আবার আটকে আছে!
    return Ok(new { user, orders });
} // এই সময় অন্য request সার্ভ হতে পারছে না
// ✅ Async — Thread মুক্ত থাকে
public async Task<IActionResult> GetUser(int id) {
    var user = await _db.Users.FindAsync(id);   // Thread মুক্ত!
    var orders = await _orderService.GetOrdersAsync(id);
    return Ok(new { user, orders });
} // await-এর সময় Thread অন্য request নিতে পারে

// ❌ Library-তে Context ক্যাপচার হয় — Deadlock ঝুঁকি
public async Task<string> FetchDataAsync() {
    var result = await httpClient.GetStringAsync(url);
    // SynchronizationContext ক্যাপচার করে
    // UI thread-এ ফিরতে চায় — Deadlock!
    return result;
}
// ✅ Library-তে ConfigureAwait(false) ব্যবহার করুন
public async Task<string> FetchDataAsync() {
    var result = await httpClient
        .GetStringAsync(url)
        .ConfigureAwait(false); // Context ক্যাপচার করবে না
    // যেকোনো Thread-এ চলবে — নিরাপদ
    return result;
}

// ✅ try/catch দিয়ে async exception handle করুন
public async Task ProcessOrdersAsync() {
    try {
        await Task.WhenAll(
            SendEmailAsync(),
            UpdateInventoryAsync(),
            GenerateReportAsync()
        );
    }
    catch (AggregateException ex) {
        // WhenAll — সব Exception একসাথে ধরা পড়ে
        foreach (var inner in ex.InnerExceptions)
            LogError(inner.Message);
    }
    catch (OperationCanceledException) {
        // CancellationToken দিয়ে বাতিল হলে
    }
}

// Task — সবসময় Heap-এ object তৈরি করে
public async Task<int> GetCachedValue() {
    if (_cache.TryGetValue("key", out var val))
        return val; // Cache hit — কিন্তু Task object তৈরি!
    return await _db.GetValueAsync();
}
// ValueTask — cache hit-এ কোনো allocation নেই!
public ValueTask<int> GetCachedValue() {
    if (_cache.TryGetValue("key", out var val))
        return new ValueTask<int>(val); // ✅ No heap alloc
    return new ValueTask<int>(_db.GetValueAsync());
}

Best Practices

Async সেরা অভ্যাস

✅
Async All the Way
async method chain ভাঙবেন না। মাঝে sync করলে Deadlock।
💡 Controller → Service → Repository — সব layer-এ async রাখুন। একটি layer sync হলে পুরো chain ভেঙে যায়।
🚫
.Result ব্যবহার নয়
.Result বা .Wait() Deadlock তৈরি করে। সবসময় await করুন।
💡 ASP.NET Core-এ ConfigureAwait(false) default, কিন্তু .Result এখনও বিপজ্জনক — UI thread capture হলে freeze।
⚡
ConfigureAwait(false)
Library কোডে সবসময় ConfigureAwait(false) লাগান।
💡 Application code (Controller) — ConfigureAwait(false) লাগানোর দরকার নেই। Library/NuGet package-এ লাগান।
🎟️
CancellationToken
সব async method-এ CancellationToken parameter রাখুন।
💡 HttpContext.RequestAborted ASP.NET Core-এ automatic cancellation token — long-running task-এ pass করুন।
